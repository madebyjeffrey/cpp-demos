  # Apple distributes no P1689 scanner. This small, explicit module graph is ordered

  string(TOUPPER "${CMAKE_BUILD_TYPE}" _scars_config_upper)
  separate_arguments(_scars_flags NATIVE_COMMAND "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${_scars_config_upper}}")
  list(APPEND _scars_flags -std=c++23 -fcxx-modules -arch "${SKIA_WINDOW_ARCH}"
    "-mmacosx-version-min=${CMAKE_OSX_DEPLOYMENT_TARGET}" -Wall -Wextra -Wpedantic)
  if(CMAKE_OSX_SYSROOT)
    list(APPEND _scars_flags -isysroot "${CMAKE_OSX_SYSROOT}")
  endif()
  set(_scars_native_flags -I "${PROJECT_SOURCE_DIR}/include")
  set(_scars_skia_flags -isystem "${SKIA_SOURCE_DIR}" -DSK_GANESH -DSK_METAL -DSK_DISABLE_TRACING -DSK_GAMMA_APPLY_TO_A8)
  add_custom_command(OUTPUT "${_scars_bmi}/scars.pcm"
    COMMAND "${CMAKE_CXX_COMPILER}" ${_scars_flags} --precompile "${_scars_interfaces}" -MD -MF "${_scars_bmi}/scars.interface.d" -o "${_scars_bmi}/scars.pcm"

endif()
