cmake_minimum_required(VERSION 3.25)
project(skia_window VERSION 0.1.0 LANGUAGES CXX)

if(NOT APPLE OR NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  message(FATAL_ERROR "skia_window currently implements macOS 26+ only.")
endif()
enable_language(OBJCXX)
if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0" CACHE STRING "Minimum macOS version" FORCE)
endif()
if(CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS "26.0")
  message(FATAL_ERROR "The minimum supported macOS deployment target is 26.0")
endif()
if(CMAKE_CONFIGURATION_TYPES)
  message(FATAL_ERROR "Use a single-configuration generator (Ninja) to match the Skia build.")
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|RelWithDebInfo)$")
  message(FATAL_ERROR "Choose Debug or RelWithDebInfo, or use the supplied presets.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 23)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(SKIA_WINDOW_ARCH "${CMAKE_OSX_ARCHITECTURES}")
if(NOT SKIA_WINDOW_ARCH)
  set(SKIA_WINDOW_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
endif()
if(NOT SKIA_WINDOW_ARCH MATCHES "^(arm64|x86_64)$")
  message(FATAL_ERROR "Build one architecture at a time: arm64 or x86_64.")
endif()
set(SKIA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/skia" CACHE PATH "Pinned Skia source")
set(SKIA_LIBRARY_DIR "${SKIA_SOURCE_DIR}/out/${SKIA_WINDOW_ARCH}-${CMAKE_BUILD_TYPE}" CACHE PATH "Matching Skia build")
include(cmake/Skia.cmake)

add_library(skia_window STATIC src/menu.cpp src/ui.cpp src/macos/backend.mm)
add_library(skia_window::skia_window ALIAS skia_window)
target_include_directories(skia_window PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(skia_window PUBLIC cxx_std_23)
target_compile_options(skia_window PRIVATE -Wall -Wextra -Wpedantic
  "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_link_libraries(skia_window PUBLIC skia_window_skia)
find_library(SKIA_WINDOW_CARBON_FRAMEWORK Carbon REQUIRED)
target_link_libraries(skia_window PRIVATE "${SKIA_WINDOW_CARBON_FRAMEWORK}")

option(SKIA_WINDOW_BUILD_DEMO "Build the interactive gallery" ${PROJECT_IS_TOP_LEVEL})
option(SKIA_WINDOW_BUILD_TESTS "Build tests (GPU tests require a logged-in desktop)" ${PROJECT_IS_TOP_LEVEL})
option(SCARS_BUILD "Build the independent module-based SCARS engine and lab" ${PROJECT_IS_TOP_LEVEL})
if(SKIA_WINDOW_BUILD_DEMO)
  add_executable(skia_window_demo MACOSX_BUNDLE demo/main.cpp demo/controls.cpp)
  target_link_libraries(skia_window_demo PRIVATE skia_window::skia_window)
  target_compile_options(skia_window_demo PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(skia_window_demo PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER "dev.skia-window.gallery"
    MACOSX_BUNDLE_BUNDLE_NAME "Skia Gallery"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/demo/Info.plist.in")
  target_sources(skia_window_demo PRIVATE demo/assets/Roboto-Regular.ttf demo/assets/texture.png demo/assets/Roboto-LICENSE.txt)
  set_source_files_properties(demo/assets/Roboto-Regular.ttf demo/assets/texture.png demo/assets/Roboto-LICENSE.txt
    PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
endif()
if(SKIA_WINDOW_BUILD_TESTS)
  enable_testing()
  add_executable(skia_window_tests tests/integration.mm)
  target_link_libraries(skia_window_tests PRIVATE skia_window::skia_window)
  target_compile_options(skia_window_tests PRIVATE -Wall -Wextra -fobjc-arc)
  add_test(NAME skia_window.integration COMMAND skia_window_tests)
  set_tests_properties(skia_window.integration PROPERTIES TIMEOUT 45 LABELS "gpu;desktop")
  add_executable(skia_window_ui_tests tests/ui.cpp)
  target_link_libraries(skia_window_ui_tests PRIVATE skia_window::skia_window)
  add_test(NAME skia_window.ui COMMAND skia_window_ui_tests)
  set_tests_properties(skia_window.ui PROPERTIES TIMEOUT 20 LABELS "ui")
  add_executable(skia_window_ui_performance_tests tests/ui_performance.cpp)
  target_link_libraries(skia_window_ui_performance_tests PRIVATE skia_window::skia_window)
  add_test(NAME skia_window.ui_performance COMMAND skia_window_ui_performance_tests)
  set_tests_properties(skia_window.ui_performance PROPERTIES TIMEOUT 20 LABELS "ui;performance")
  add_executable(skia_window_resize_tests tests/resize.mm)
  target_link_libraries(skia_window_resize_tests PRIVATE skia_window::skia_window)
  target_compile_options(skia_window_resize_tests PRIVATE -Wall -Wextra -fobjc-arc)
  add_test(NAME skia_window.resize COMMAND skia_window_resize_tests)
  set_tests_properties(skia_window.resize PROPERTIES TIMEOUT 20 LABELS "ui;gpu;desktop;performance")
  add_executable(skia_window_ui_desktop_tests tests/ui_desktop.mm)
  target_link_libraries(skia_window_ui_desktop_tests PRIVATE skia_window::skia_window)
  target_compile_options(skia_window_ui_desktop_tests PRIVATE -Wall -Wextra -fobjc-arc)
  add_test(NAME skia_window.ui_desktop COMMAND skia_window_ui_desktop_tests)
  set_tests_properties(skia_window.ui_desktop PROPERTIES TIMEOUT 30 LABELS "ui;gpu;desktop")
  add_executable(skia_window_ui_gpu_tests tests/ui_gpu.mm)
  target_link_libraries(skia_window_ui_gpu_tests PRIVATE skia_window::skia_window)
  target_compile_options(skia_window_ui_gpu_tests PRIVATE -Wall -Wextra -fobjc-arc)
  add_test(NAME skia_window.ui_gpu COMMAND skia_window_ui_gpu_tests)
  set_tests_properties(skia_window.ui_gpu PROPERTIES TIMEOUT 30 LABELS "ui;gpu")
  add_test(NAME skia_window.ui_native_actions COMMAND skia_window_ui_desktop_tests --actions-only)
  set_tests_properties(skia_window.ui_native_actions PROPERTIES TIMEOUT 30 LABELS "ui;gpu;native")
  add_subdirectory(tests/consumer)
endif()
if(SCARS_BUILD)
  add_subdirectory(scars)
endif()
